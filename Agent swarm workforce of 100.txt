// --- SUB-COMPONENTS ---

const AgentSwarm = ({ isActive }) => {
  // Generate coordinates for 100 agents
  const agents = useMemo(() => {
    return Array.from({ length: 100 }).map((_, i) => ({
      id: i,
      x: Math.random() * 100,
      y: Math.random() * 100,
      size: Math.random() * 1.5 + 0.5,
      delay: Math.random() * 5,
      duration: 3 + Math.random() * 4,
    }));
  }, []);

  return (
    <div className="relative w-full h-full overflow-hidden bg-zinc-950 rounded-[40px] border border-white/10 shadow-2xl transition-all duration-1000">
      <div className="absolute inset-0 opacity-10" 
           style={{ backgroundImage: 'radial-gradient(circle, #3b82f6 1px, transparent 1px)', backgroundSize: '40px 40px' }} />
      
      <svg viewBox="0 0 100 100" className="w-full h-full">
        <defs>
          <filter id="glowAgent">
            <feGaussianBlur stdDeviation="0.5" result="coloredBlur"/>
            <feMerge>
              <feMergeNode in="coloredBlur"/><feMergeNode in="SourceGraphic"/>
            </feMerge>
          </filter>
        </defs>
        {agents.map((agent) => (
          <g key={agent.id}>
            <circle 
              cx={agent.x} 
              cy={agent.y} 
              r={agent.size * 0.3} 
              fill={isActive ? '#60a5fa' : '#3f3f46'}
              filter={isActive ? "url(#glowAgent)" : ""}
              className="transition-colors duration-1000"
            >
              <animate 
                attributeName="opacity" 
                values="0.2;1;0.2" 
                dur={`${agent.duration}s`} 
                begin={`${agent.delay}s`}
                repeatCount="indefinite" 
              />
              {isActive && (
                <animateTransform
                  attributeName="transform"
                  type="translate"
                  values="0 0; 1 1; 0 0"
                  dur={`${agent.duration * 2}s`}
                  repeatCount="indefinite"
                  additive="sum"
                />
              )}
            </circle>
          </g>
        ))}
      </svg>

      <div className={`absolute bottom-8 left-8 right-8 flex justify-between items-end transition-opacity duration-1000 ${isActive ? 'opacity-100' : 'opacity-20'}`}>
        <div className="space-y-1">
          <div className="text-[8px] font-mono text-blue-400 uppercase tracking-[0.3em]">Neural Workforce</div>
          <div className="text-xs font-bold text-white uppercase tracking-tighter">100 Active Agents</div>
        </div>
        <div className="flex gap-6">
          <div className="text-right">
            <div className="text-[8px] font-mono text-zinc-500 uppercase tracking-widest">Efficiency</div>
            <div className="text-xs font-bold text-white tracking-tighter">99.8%</div>
          </div>
          <div className="text-right">
            <div className="text-[8px] font-mono text-zinc-500 uppercase tracking-widest">Compute</div>
            <div className="text-xs font-bold text-white italic tracking-tighter">CAMBER_OS</div>
          </div>
        </div>
      </div>
    </div>
  );
};
