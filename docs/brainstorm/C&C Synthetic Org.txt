// --- THE NEURAL CONSTELLATION (IMPRESSIVE AGENT DISPLAY) ---
const NeuralConstellation = ({ activeDept }) => {
  const departments = [
    { id: 'research', color: '#60a5fa', center: { x: 30, y: 35 }, label: 'Research' },
    { id: 'finance', color: '#34d399', center: { x: 70, y: 35 }, label: 'Finance' },
    { id: 'sales', color: '#fbbf24', center: { x: 30, y: 65 }, label: 'Sales' },
    { id: 'security', color: '#f87171', center: { x: 70, y: 65 }, label: 'Security' },
  ];

  const agents = useMemo(() => {
    return Array.from({ length: 100 }).map((_, i) => {
      const deptIdx = i % 4;
      const dept = departments[deptIdx];
      return {
        id: i,
        deptId: dept.id,
        // Cluster agents around their department center
        baseX: dept.center.x + (Math.random() - 0.5) * 20,
        baseY: dept.center.y + (Math.random() - 0.5) * 20,
        size: Math.random() * 0.8 + 0.4,
        speed: 2 + Math.random() * 3,
        offset: Math.random() * Math.PI * 2
      };
    });
  }, []);

  return (
    <div className="relative w-full h-full bg-[#050505] rounded-[48px] border border-white/5 shadow-2xl overflow-hidden group">
      {/* Structural Grid */}
      <div className="absolute inset-0 opacity-5" 
           style={{ backgroundImage: 'radial-gradient(circle, #fff 1px, transparent 1px)', backgroundSize: '30px 30px' }} />
      
      <svg viewBox="0 0 100 100" className="w-full h-full relative z-10">
        <defs>
          <filter id="hyperGlow" x="-50%" y="-50%" width="200%" height="200%">
            <feGaussianBlur stdDeviation="1.2" result="blur" />
            <feComposite in="SourceGraphic" in2="blur" operator="over" />
          </filter>
        </defs>

        {/* Dynamic Connection Web */}
        <g opacity="0.15">
          {departments.map(dept => (
            agents.filter(a => a.deptId === dept.id && a.id % 5 === 0).map((a, i) => (
              <line 
                key={`line-${a.id}`}
                x1={a.baseX} y1={a.baseY} 
                x2={dept.center.x} y2={dept.center.y} 
                stroke={dept.color} 
                strokeWidth="0.05"
              />
            ))
          ))}
        </g>

        {/* The 100 Agents */}
        {agents.map((agent) => {
          const isHighlighted = activeDept === agent.deptId || !activeDept;
          const dept = departments.find(d => d.id === agent.deptId);
          
          return (
            <g key={agent.id}>
              <circle 
                cx={agent.baseX} 
                cy={agent.baseY} 
                r={agent.size * (isHighlighted ? 0.6 : 0.3)} 
                fill={isHighlighted ? dept.color : '#27272a'}
                filter={isHighlighted ? "url(#hyperGlow)" : ""}
                className="transition-all duration-700 ease-out"
              >
                <animate 
                  attributeName="opacity" 
                  values="0.4;1;0.4" 
                  dur={`${agent.speed}s`} 
                  repeatCount="indefinite" 
                />
              </circle>
            </g>
          );
        })}

        {/* Department Labels in SVG space */}
        {departments.map(dept => (
          <text 
            key={dept.id}
            x={dept.center.x} 
            y={dept.center.y - 12} 
            textAnchor="middle" 
            className={`text-[3px] font-bold uppercase tracking-[0.2em] transition-all duration-500 fill-white 
              ${activeDept === dept.id ? 'opacity-100 scale-110' : 'opacity-20'}`}
          >
            {dept.label}
          </text>
        ))}
      </svg>

      {/* Interface Overlays */}
      <div className="absolute top-8 left-8 flex flex-col gap-2">
        <div className="bg-white/5 backdrop-blur-md border border-white/10 px-4 py-2 rounded-full flex items-center gap-3">
          <div className="w-2 h-2 bg-blue-500 rounded-full animate-pulse" />
          <span className="text-[10px] font-mono text-white/60 tracking-widest uppercase">Autonomous Org_Sync: Active</span>
        </div>
      </div>

      <div className="absolute bottom-8 right-8 text-right">
        <div className="text-[9px] font-mono text-zinc-500 uppercase tracking-[0.4em] mb-1">Compute Layer</div>
        <div className="text-sm font-bold text-white tracking-tighter italic font-serif">Ironbark v4.02</div>
      </div>
    </div>
  );
};
