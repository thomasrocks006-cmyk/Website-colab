// --- AGENT SWARM VISUALIZER (ENHANCED) ---
const AgentSwarm = ({ isActive }) => {
  const agents = useMemo(() => {
    return Array.from({ length: 100 }).map((_, i) => ({
      id: i,
      x: Math.random() * 100,
      y: Math.random() * 100,
      size: Math.random() * 1.5 + 0.8,
      group: i % 4, // 0: Strategy, 1: Data, 2: Security, 3: Logic
      pulseDelay: Math.random() * 5,
    }));
  }, []);

  const getGroupColor = (group) => {
    if (!isActive) return '#27272a'; // zinc-800
    switch(group) {
      case 0: return '#60a5fa'; // blue
      case 1: return '#818cf8'; // indigo
      case 2: return '#34d399'; // emerald
      default: return '#f472b6'; // pink
    }
  };

  return (
    <div className="relative w-full h-full overflow-hidden bg-[#09090b] rounded-[48px] border border-white/5 shadow-2xl transition-all duration-1000">
      {/* Background Matrix Grid */}
      <div className="absolute inset-0 opacity-[0.15]" 
           style={{ 
             backgroundImage: `linear-gradient(to right, #27272a 1px, transparent 1px), linear-gradient(to bottom, #27272a 1px, transparent 1px)`,
             backgroundSize: '40px 40px' 
           }} />
      
      <svg viewBox="0 0 100 100" className="w-full h-full relative z-10">
        <defs>
          <filter id="agentGlow" x="-50%" y="-50%" width="200%" height="200%">
            <feGaussianBlur stdDeviation="0.8" result="blur" />
            <feComposite in="SourceGraphic" in2="blur" operator="over" />
          </filter>
        </defs>

        {/* Dynamic Connection Lines (Only active for a subset to prevent clutter) */}
        {isActive && agents.filter(a => a.id % 8 === 0).map((agent, i) => (
          <line 
            key={`line-${i}`}
            x1={agent.x} y1={agent.y} 
            x2="50" y2="50" 
            stroke={getGroupColor(agent.group)}
            strokeWidth="0.08"
            opacity="0.2"
            className="animate-pulse"
          />
        ))}

        {/* 100 Individual Agents */}
        {agents.map((agent) => (
          <g key={agent.id} className="transition-all duration-1000">
            <circle 
              cx={agent.x} 
              cy={agent.y} 
              r={agent.size * 0.4} 
              fill={getGroupColor(agent.group)}
              filter={isActive ? "url(#agentGlow)" : ""}
              className="transition-all duration-700"
            >
              <animate 
                attributeName="opacity" 
                values="0.2;1;0.2" 
                dur={`${2 + agent.pulseDelay}s`} 
                begin={`${agent.pulseDelay}s`}
                repeatCount="indefinite" 
              />
              {isActive && (
                <animateTransform
                  attributeName="transform"
                  type="translate"
                  values={`${Math.sin(agent.id)} ${Math.cos(agent.id)}; ${Math.cos(agent.id)} ${Math.sin(agent.id)}; ${Math.sin(agent.id)} ${Math.cos(agent.id)}`}
                  dur="10s"
                  repeatCount="indefinite"
                  additive="sum"
                />
              )}
            </circle>
          </g>
        ))}

        {/* Central Command Core */}
        <circle cx="50" cy="50" r="1.5" fill="white" className={isActive ? 'animate-ping' : ''} />
      </svg>

      {/* Real-time Status Board */}
      <div className={`absolute top-10 left-10 space-y-2 transition-all duration-1000 ${isActive ? 'opacity-100 translate-x-0' : 'opacity-0 -translate-x-10'}`}>
        <div className="flex items-center gap-3 text-[10px] font-mono text-zinc-500 uppercase tracking-widest bg-black/40 backdrop-blur-md px-3 py-1.5 rounded-full border border-white/5">
          <Activity size={10} className="text-blue-500" /> System_Core: Active
        </div>
        <div className="flex items-center gap-3 text-[10px] font-mono text-zinc-500 uppercase tracking-widest bg-black/40 backdrop-blur-md px-3 py-1.5 rounded-full border border-white/5">
          <Fingerprint size={10} className="text-emerald-500" /> Identity_Nodes: 100/100
        </div>
      </div>

      <div className={`absolute bottom-10 left-10 right-10 flex justify-between items-end transition-opacity duration-1000 ${isActive ? 'opacity-100' : 'opacity-20'}`}>
        <div className="space-y-1">
          <div className="text-[10px] font-mono text-blue-400 uppercase tracking-[0.4em] font-bold">The Synthesis Workforce</div>
          <div className="text-sm font-bold text-white uppercase tracking-tighter">Autonomous Organization Level: 04</div>
        </div>
        <div className="hidden md:flex gap-10">
          <div className="text-right">
            <div className="text-[9px] font-mono text-zinc-500 uppercase tracking-widest">Logic Density</div>
            <div className="text-sm font-bold text-white tracking-tighter">High</div>
          </div>
          <div className="text-right">
            <div className="text-[9px] font-mono text-zinc-500 uppercase tracking-widest">Foundation</div>
            <div className="text-sm font-bold text-white italic tracking-tighter">IRONBARK_v4</div>
          </div>
        </div>
      </div>
    </div>
  );
};
