// --- AGENT HUB VISUALIZER ---
const WorkforceEngine = ({ activeRole }) => {
  const [pulse, setPulse] = useState(0);

  useEffect(() => {
    const interval = setInterval(() => setPulse(p => (p + 1) % 100), 50);
    return () => clearInterval(interval);
  }, []);

  // 100 Agents organized into 4 hubs
  const hubs = [
    { id: 0, label: "Hustlers", color: "#3b82f6", pos: { x: 30, y: 35 } },
    { id: 1, label: "Money", color: "#10b981", pos: { x: 70, y: 35 } },
    { id: 2, label: "Admin", color: "#f59e0b", pos: { x: 30, y: 65 } },
    { id: 3, label: "Watchdogs", color: "#ef4444", pos: { x: 70, y: 65 } }
  ];

  const agents = useMemo(() => {
    return Array.from({ length: 100 }).map((_, i) => ({
      id: i,
      hub: i % 4,
      offset: { x: (Math.random() - 0.5) * 25, y: (Math.random() - 0.5) * 25 },
      size: 0.5 + Math.random() * 1
    }));
  }, []);

  return (
    <div className="relative w-full h-full bg-[#050505] rounded-[40px] border border-white/5 overflow-hidden shadow-2xl group">
      {/* Background Grid */}
      <div className="absolute inset-0 opacity-[0.03]" style={{ backgroundImage: 'radial-gradient(#fff 1px, transparent 1px)', backgroundSize: '32px 32px' }} />
      
      <svg viewBox="0 0 100 100" className="w-full h-full relative z-10 p-10">
        <defs>
          <filter id="glow">
            <feGaussianBlur stdDeviation="1.5" result="blur" />
            <feComposite in="SourceGraphic" in2="blur" operator="over" />
          </filter>
        </defs>

        {/* Connection Lines between Hubs */}
        <g opacity="0.1">
          <line x1="30" y1="35" x2="70" y2="35" stroke="white" strokeWidth="0.1" />
          <line x1="70" y1="35" x2="70" y2="65" stroke="white" strokeWidth="0.1" />
          <line x1="70" y1="65" x2="30" y2="65" stroke="white" strokeWidth="0.1" />
          <line x1="30" y1="65" x2="30" y2="35" stroke="white" strokeWidth="0.1" />
        </g>

        {/* Individual Agents */}
        {agents.map((agent) => {
          const hub = hubs[agent.hub];
          const isActive = activeRole === null || activeRole === agent.hub;
          return (
            <circle 
              key={agent.id}
              cx={hub.pos.x + agent.offset.x}
              cy={hub.pos.y + agent.offset.y}
              r={isActive ? agent.size * 0.4 : agent.size * 0.2}
              fill={isActive ? hub.color : '#27272a'}
              filter={isActive ? "url(#glow)" : ""}
              opacity={isActive ? 0.8 : 0.1}
              className="transition-all duration-700 ease-in-out"
            >
              <animate attributeName="opacity" values="0.4;1;0.4" dur={`${2 + (agent.id % 3)}s`} repeatCount="indefinite" />
            </circle>
          );
        })}

        {/* Hub Centers */}
        {hubs.map(hub => (
          <g key={hub.id} className="transition-opacity duration-500" opacity={activeRole === null || activeRole === hub.id ? 1 : 0.2}>
            <text x={hub.pos.x} y={hub.pos.y - 15} textAnchor="middle" className="fill-white/20 text-[3px] font-bold tracking-[0.3em] uppercase">{hub.label}</text>
            <circle cx={hub.pos.x} cy={hub.pos.y} r="1.5" fill={hub.color} className="animate-pulse" />
          </g>
        ))}
      </svg>

      {/* Real-time Telemetry */}
      <div className="absolute bottom-10 left-10 right-10 flex justify-between items-end">
        <div className="space-y-1">
          <div className="text-[8px] font-mono text-white/20 uppercase tracking-[0.4em]">Active Status</div>
          <div className="text-xs font-bold text-white uppercase tracking-tighter">
            {activeRole !== null ? `${hubs[activeRole].label} Engaged` : "100 Agents Standing By"}
          </div>
        </div>
        <div className="flex gap-6 text-right">
          <div>
            <div className="text-[8px] font-mono text-zinc-600 uppercase tracking-widest">Load</div>
            <div className="text-xs font-bold text-white">Optimal</div>
          </div>
          <div>
            <div className="text-[8px] font-mono text-zinc-600 uppercase tracking-widest">Core</div>
            <div className="text-xs font-bold text-white italic">Camber_v4</div>
          </div>
        </div>
      </div>
    </div>
  );
};
